class ES2CM(nn.Module):
    """
    增强RGB特征提取模块 (Enhanced RGB Spectral-Spatial Context Module)
    专门处理3通道RGB数据，提取瘀伤相关的基础特征
    """

    def __init__(self, rgb_channels=3, output_channels=48, hidden_dim=32,
                 use_residual=True, use_attention=True):
        super().__init__()
        self.output_channels = output_channels
        self.use_residual = use_residual
        self.use_attention = use_attention

        # 1. 初始投影：将3通道RGB扩展到更高维度
        self.initial_projection = nn.Sequential(
            nn.Conv2d(rgb_channels, hidden_dim, 3, padding=1),
            nn.BatchNorm2d(hidden_dim),
            nn.ReLU(inplace=True),

            # 进一步扩大维度，增强特征表达能力
            nn.Conv2d(hidden_dim, hidden_dim * 2, 3, padding=1),
            nn.BatchNorm2d(hidden_dim * 2),
            nn.ReLU(inplace=True),
        )

        # 2. 多尺度特征提取（参考ResNet的bottleneck设计）
        self.multiscale_blocks = nn.ModuleList([
            # 尺度1: 3x3卷积捕获局部细节
            nn.Sequential(
                nn.Conv2d(hidden_dim * 2, hidden_dim, 3, padding=1, dilation=1),
                nn.BatchNorm2d(hidden_dim),
                nn.ReLU(inplace=True),
            ),
            # 尺度2: 5x5卷积捕获中等范围特征
            nn.Sequential(
                nn.Conv2d(hidden_dim * 2, hidden_dim, 5, padding=2, dilation=1),
                nn.BatchNorm2d(hidden_dim),
                nn.ReLU(inplace=True),
            ),
            # 尺度3: 3x3空洞卷积捕获更大感受野
            nn.Sequential(
                nn.Conv2d(hidden_dim * 2, hidden_dim, 3, padding=2, dilation=2),
                nn.BatchNorm2d(hidden_dim),
                nn.ReLU(inplace=True),
            )
        ])

        # 3. 特征融合层（参考MobileNet的深度可分离卷积思想）
        self.feature_fusion = nn.Sequential(
            # 深度卷积处理空间信息
            nn.Conv2d(hidden_dim * 3, hidden_dim * 3, 3, padding=1, groups=hidden_dim * 3),
            nn.BatchNorm2d(hidden_dim * 3),
            nn.ReLU(inplace=True),

            # 点卷积整合通道信息
            nn.Conv2d(hidden_dim * 3, hidden_dim * 2, 1),
            nn.BatchNorm2d(hidden_dim * 2),
            nn.ReLU(inplace=True),
        )

        # 4. 瘀伤特异性注意力机制（针对瘀伤颜色特征优化）
        if self.use_attention:
            self.bruise_attention = nn.Sequential(
                # 颜色统计注意力：瘀伤通常有特定的颜色分布
                nn.AdaptiveAvgPool2d(1),
                nn.Conv2d(hidden_dim * 2, hidden_dim // 4, 1),
                nn.ReLU(inplace=True),
                nn.Conv2d(hidden_dim // 4, hidden_dim * 2, 1),
                nn.Sigmoid(),

                # 空间显著性注意力：瘀伤区域的空间连续性
                nn.Conv2d(1, 1, 7, padding=3),  # 使用较大卷积核增强空间连续性
                nn.Sigmoid()
            )

        # 5. 残差连接增强（参考ResNet思想）
        if self.use_residual:
            self.residual_adapter = nn.Sequential(
                nn.Conv2d(hidden_dim * 2, hidden_dim * 2, 1),
                nn.BatchNorm2d(hidden_dim * 2),
            )

        # 6. 最终投影到目标输出维度
        self.final_projection = nn.Sequential(
            nn.Conv2d(hidden_dim * 2, output_channels, 3, padding=1),
            nn.BatchNorm2d(output_channels),
            nn.ReLU(inplace=True),

            # 可选：添加dropout防止过拟合
            nn.Dropout2d(0.1),
        )

        # 初始化权重
        self._init_weights()

    def _init_weights(self):
        """权重初始化，确保训练稳定性"""
        for m in self.modules():
            if isinstance(m, nn.Conv2d):
                nn.init.kaiming_normal_(m.weight, mode='fan_out', nonlinearity='relu')
                if m.bias is not None:
                    nn.init.constant_(m.bias, 0)
            elif isinstance(m, nn.BatchNorm2d):
                nn.init.constant_(m.weight, 1)
                nn.init.constant_(m.bias, 0)

    def forward(self, rgb_input):
        """
        Args:
            rgb_input: 3通道RGB输入 [B, 3, H, W]

        Returns:
            output: 提取的特征 [B, output_channels, H, W]
            attention_map: 注意力图（用于可视化）[B, 1, H, W]
        """
        # 1. 初始投影和维度扩展
        x = self.initial_projection(rgb_input)  # [B, 64, H, W]

        # 保存残差连接
        if self.use_residual:
            residual = self.residual_adapter(x) if hasattr(self, 'residual_adapter') else x

        # 2. 多尺度特征提取
        scale_features = []
        for block in self.multiscale_blocks:
            scale_feat = block(x)
            scale_features.append(scale_feat)

        # 拼接多尺度特征
        multiscale_fused = torch.cat(scale_features, dim=1)  # [B, 96, H, W]

        # 3. 特征融合
        fused = self.feature_fusion(multiscale_fused)  # [B, 64, H, W]

        # 4. 应用残差连接
        if self.use_residual:
            fused = fused + residual

        # 5. 瘀伤特异性注意力
        attention_map = None
        if self.use_attention:
            # 生成通道注意力
            channel_attn = self.bruise_attention[:4](fused)  # [B, 64, 1, 1]
            attended = fused * channel_attn

            # 生成空间注意力（基于特征强度）
            spatial_attn = torch.mean(attended, dim=1, keepdim=True)  # [B, 1, H, W]
            spatial_attn = self.bruise_attention[4](spatial_attn)  # [B, 1, H, W]

            attended = attended * spatial_attn
            attention_map = spatial_attn
        else:
            attended = fused

        # 6. 最终投影到目标维度
        output = self.final_projection(attended)  # [B, output_channels, H, W]

        return output
